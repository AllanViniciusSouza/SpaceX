VAR_INPUT
   ID : HW_IO;   // The Profinet Device Hardware Identifier under device properties
END_VAR

VAR_OUTPUT
   Done : Bool;   // Read done successfully
   Error : Bool;   // Error during reading process
   GrossWeight : Real;
   TareWeight : Real;
   NetWeight : Real;
   TareMode : SInt;   // 0,tare;1;preset tare
   Unit : SInt;   // Unit: 0-g; 1-kg, 2-lb
END_VAR

VAR_IN_OUT
   ePrint : Bool;   // Toggle this bit to execute an ePrint from terminal > PLC
   RecordNumber : DInt;   // record number which is to be read out from the weighing indicator
   Get_TransactionNum : Bool;   // Trigger this bit to get the previously read alibi record number.
   Set_TransactionNum : Bool;   // Trigger this bit to set the alibi record number which is to be read out
   Get_Alibi : Bool;   // Trigger this bit to read out the alibi transaction record specified by "RecordNum".
   Alibi_Array : Array[0..44] of Byte;
END_VAR

VAR
   RDREC01 : RDREC;
   RDREC02 : RDREC;
   WRREC01 : WRREC;
   WRREC02 : WRREC;
   UINT : SInt;
END_VAR
