VAR_INPUT
   ID : HW_IO;   // The Profinet Device Hardware Identifier under device properties
   ScaleCapacity : Real;   // Scale capacity to be written into the transmitter
   Increment : Real;   // Increment value to be written into the transmitter
END_VAR

VAR_OUTPUT
   Done : Bool;   // Write process done successfully
   Error : Bool;   // Error during writing process
END_VAR

VAR_IN_OUT
   Write : Bool;   // Switch this bit ON to start writing
END_VAR

VAR
   Static01 : Bool;
   Step : Int;
   WRREC_CAP : WRREC;
   CAP_RECORD : Real;
   INC_RECORD : Real;
   Activate : Bool;
END_VAR

   operands: #Write, #Static01, #ScaleCapacity, #Increment, #Error, #Done, #Activate, #Step
